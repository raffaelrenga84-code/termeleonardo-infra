# POS di Bistrot e ristorante, fase 1 (comande) — piano di lavoro

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** un POS sui palmari Sunmi che prende le comande a portate, con note libere, preferiti, varianti e prezzi, stampa in cucina e al bar, funziona senza internet grazie al PC del Bistrot e ha il cloud di riserva; raggiungibile su `https://www.hoteltermeleonardo.com/pos` solo dall'IP dell'hotel.

**Architecture:** tre pezzi con lo stesso contratto HTTP (`?a=<azione>`): la pagina `pagine/pos/` (web app installabile), la funzione Supabase `pos` (Postgres, tabelle `pos_*`), il server locale `pos-locale/` (Deno + SQLite sul PC del Bistrot, HTTPS con CA privata, stampa ESC/POS sulla LAN, allineamento col cloud). Le regole stanno in moduli puri provati con Deno; la pagina e la funzione sono letti dal sorgente dalle prove, come nel resto del repo.

**Tech Stack:** Deno 2.9 (funzione e server locale, `node:sqlite` per SQLite), Supabase (Postgres, Edge Functions, secrets), HTML/JS senza framework come le altre pagine, ESC/POS su TCP 9100, Vercel (riscrittura con condizione `x-real-ip`), estensione Chrome (`estensione/`) per importare gli articoli da Fidra, `openssl` per la CA privata.

// supabase/functions/pos/portate.ts — l'ordine delle portate e la regola dell'invio.
// «Antipasto prima; poi, dopo 15 minuti o quando decide il cameriere, la
// pasta» (la proprieta', 4 settembre 2026): all'invio escono subito le
// bevande e la portata di cibo piu' bassa fra quelle da inviare; le altre
// restano in attesa e partono con «Vai», una per volta, in ordine.
export const PORTATE = ['bevande', 'antipasti', 'primi', 'secondi', 'dolci'] as const;
export type Portata = typeof PORTATE[number];
type Riga = { portata: Portata; stato: string };

export const ordine = (p: Portata): number => PORTATE.indexOf(p);

export function dividi(righe: Riga[]): { subito: Portata[]; attesa: Portata[] } {
  const daInviare = [...new Set(righe.filter((r) => r.stato === 'da_inviare').map((r) => r.portata))]
    .sort((a, b) => ordine(a) - ordine(b));
  const cibo = daInviare.filter((p) => p !== 'bevande');
  const subito: Portata[] = [];
  if (daInviare.includes('bevande')) subito.push('bevande');
  if (cibo.length) subito.push(cibo[0]);
  return { subito, attesa: cibo.slice(1) };
}

export function prossima(righe: Riga[]): Portata | null {
  const inAttesa = righe.filter((r) => r.stato === 'inviata').map((r) => r.portata).sort((a, b) => ordine(a) - ordine(b));
  return inAttesa[0] ?? null;
}
```

// supabase/functions/pos/conto.ts — il prezzo di una riga e il totale di un conto. Tutto in centesimi.
type Articolo = { prezzo_cent: number; prezzo_libero: boolean };
type Scelta = { articolo: Articolo; variante?: { supplemento_cent: number } | null; prezzo_manuale_cent?: number | null };

export function prezzoRiga(s: Scelta, puoPrezzoManuale: boolean): number {
  const manuale = s.prezzo_manuale_cent;
  if (s.articolo.prezzo_libero) {
    if (manuale === undefined || manuale === null) throw new Error('prezzo richiesto');
    return manuale;
  }
  if (manuale !== undefined && manuale !== null) {
    if (!puoPrezzoManuale) throw new Error('prezzo a mano non permesso');
    return manuale;
  }
  return s.articolo.prezzo_cent + (s.variante?.supplemento_cent ?? 0);
}

export function totaleCent(righe: { quantita: number; prezzo_cent: number; stato: string }[]): number {
  return righe.filter((r) => r.stato !== 'stornata').reduce((t, r) => t + r.quantita * r.prezzo_cent, 0);
}

export const euro = (cent: number): string => (cent / 100).toFixed(2).replace('.', ',') + ' €';
```

// Larghezza 32 colonne (rotolo 80 mm, carattere normale doppio per il titolo).
export type Biglietto = {
  tipo: 'COMANDA' | 'VAI' | 'STORNO' | 'MODIFICA'; locale: string; tavolo: string; conto: string; coperti: number;
  portata: string; ora: string; cameriere: string;
  righe: { quantita: number; nome: string; variante?: string | null; nota?: string | null }[];
  noteVitto?: string | null;
};
const L = 32;
const taglia = (s: string) => s.length <= L ? s : s.slice(0, L);
const aCapo = (s: string, rientro = 4): string[] => {
  const parole = s.split(' '); const out: string[] = []; let riga = '';
  for (const p of parole) {
    if ((riga + ' ' + p).trim().length > L - rientro) { out.push(riga.trim()); riga = p; } else riga = (riga + ' ' + p).trim();
  }
  if (riga) out.push(riga);
  return out.map((r) => ' '.repeat(rientro) + r);
};

export function testoBiglietto(b: Biglietto): string {
  const righe: string[] = [];
  righe.push(taglia(`${b.tipo}  ${b.portata.toUpperCase()}`));
  righe.push(taglia(`${b.tavolo}  (${b.coperti} cop.)  ${b.conto}`));
  righe.push(taglia(`${b.ora}  ${b.cameriere}  ${b.locale}`));
  if (b.noteVitto) righe.push(taglia(`!!! ${b.noteVitto} !!!`));
  righe.push('-'.repeat(L));
  for (const r of b.righe) {
    righe.push(...aCapo(`${r.quantita} x ${r.nome}`, 0));
    if (r.variante) righe.push(...aCapo(`+ ${r.variante}`));
    if (r.nota) righe.push(...aCapo(`> ${r.nota}`));
  }
  righe.push('-'.repeat(L));
  return righe.join('\n') + '\n';
}

/* CP1252 per gli accenti: la stampante (Epson-compatibile) la seleziona con ESC t 16. */
function cp1252(s: string): number[] {
  const out: number[] = [];
  for (const ch of s) {
    const c = ch.codePointAt(0)!;
    if (c < 0x80) out.push(c);
    else if (c >= 0xa0 && c <= 0xff) out.push(c);
    else if (c === 0x20ac) out.push(0x80);
    else out.push(0x3f);
  }
  return out;
}

export function escpos(testo: string): Uint8Array {
  const b: number[] = [0x1b, 0x40, 0x1b, 0x74, 16];       // ESC @, ESC t 16 (CP1252)
  const righe = testo.split('\n');
  righe.forEach((r, i) => {
    if (i === 0) b.push(0x1b, 0x21, 0x30);                  // ESC ! 0x30: doppia altezza e larghezza
    b.push(...cp1252(r), 0x0a);
    if (i === 0) b.push(0x1b, 0x21, 0x00);
  });
  b.push(0x0a, 0x0a, 0x0a, 0x1d, 0x56, 0x00);              // avanzamento e GS V 0: taglio
  return new Uint8Array(b);
}

// pagine/pos/server.js — quale server, e la coda quando nessuno risponde.
export function creaServer({ locale, cloud, ping, timeoutMs = 1000, ogniMs = 30000, adesso = Date.now }) {
  let corrente = 'cloud', ultimoTentativo = -Infinity;
  const provaLocale = async () => {
    try {
      const ok = await Promise.race([ping(locale), new Promise((r) => setTimeout(() => r(false), timeoutMs))]);
      return ok === true;
    } catch { return false; }
  };
  return {
    async base() {
      if (!locale) { corrente = 'cloud'; return cloud; }
      if (corrente === 'locale' || adesso() - ultimoTentativo >= ogniMs) {
        ultimoTentativo = adesso();
        corrente = (await provaLocale()) ? 'locale' : 'cloud';
      }
      return corrente === 'locale' ? locale : cloud;
    },
    stato() { return corrente; },
    segnaSolo() { corrente = 'solo'; },
  };
}
export function creaCoda({ salva, leggi }) {
  const eRete = (e) => e instanceof TypeError || /fetch|network|rete/i.test(String(e?.message));
  return {
    metti(richiesta) { salva([...(leggi() || []), richiesta]); },
    async svuota(chiama) {
      let coda = leggi() || [];
      while (coda.length) {
        try { await chiama(coda[0]); coda = coda.slice(1); salva(coda); }
        catch (e) { if (eRete(e)) break; coda = coda.slice(1); salva(coda); }
      }
      return { rimaste: coda.length };
    },
  };
}
